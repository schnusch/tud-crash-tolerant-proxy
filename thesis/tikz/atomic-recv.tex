\newcommand{\shortcut}[3][]{%
    \draw[->,#1] (shortcutstart) -- ({shortcutstart} |- {$(#2) + (0,-0.5)$}) -- (#2);
    \coordinate (shortcutstart) at ({shortcutstart} |- {$(#2) + (0,-0.5)$});
    \node[anchor=south west] at ({$(shortcutstart) + (0.1,0.1)$} |- {#2.north}) {#3};
}%
\newcommand{\direct}[2]{%
    \draw[->] ($(#1.south) + (5em,0)$) to[bend left] ($(#2.north) + (5em,0)$);
}%
\def\shortcutstart{switch}%
\begin{tikzpicture}[x=12em,y=-4em,>=Stealth]

\node[circle,fill,inner sep=.2em] (start)  at (0,-1.25) {};
\node[circle,draw,inner sep=.5em] (switch) at (0,-0.5) {};
\coordinate (shortcutstart) at (switch.south);
\node[draw,rounded corners,anchor=north,inner sep=.5em] (init) at (1,0) {%
    \shortstack[l]{%
        \texttt{m.msg\_len := -1}\\
        \texttt{state := ATOMIC\_DO\_SYSCALL}%
    }%
};
\node[draw,rounded corners,anchor=north,inner sep=.5em] (syscall) at ($(init.south) + (0,1)$) {%
    \shortstack[l]{%
        \texttt{m.msg\_hdr := /* ... */}\\
        \texttt{recvmmsg(sockfd, \&m, 1,}\\
        \texttt{\ \ \ \ MSG\_DONTWAIT, NULL)}%
    }%
};
\node[draw,rounded corners,anchor=north,inner sep=.5em] (done) at ($(syscall.south) + (0,1)$) {%
    \shortstack[l]{%
        \texttt{state := ATOMIC\_SWAP}\\
        \texttt{\ \ \ \ | (active << 7)}%
        \csname @ifundefined\endcsname{r@fig:atomic-swap}{}{%
            \\
            \hphantom{\texttt{state := ATOMIC\_SWAP}}%
            \llap{\footnotesize (see \cref{fig:atomic-swap})}%
        }%
    }%
};
\node[draw,rounded corners,anchor=north,inner sep=.5em] (swap) at ($(done.south) + (0,1)$) {%
    \shortstack[l]{%
        \texttt{active := state \& (1 << 7)}\\
        \texttt{ranges[!active] :=}\\
        \texttt{\ \ \ \ append\_bytes(ranges[active])}\\
        \texttt{active := !active}\\
        \texttt{state := ATOMIC\_INIT}%
    }%
};
\node[circle,draw,double,inner sep=.5em] (end) at ($(swap.south) + (5em,1)$) {};

\draw[->] (start) -- (switch);
\direct{init}{syscall}
\direct{syscall}{done}
\direct{done}{swap}
\draw[->] ($(swap.south) + (5em,0)$) to[bend left] (end);

\shortcut{init}{%
    \texttt{state == ATOMIC\_INIT}%
}
\shortcut{syscall}{%
    \shortstack[l]{%
        \texttt{state == ATOMIC\_DO\_SYSCALL}\\
        \texttt{\&\& m.msg\_len == -1}%
    }%
}
\shortcut{done}{%
    \shortstack[l]{%
        \texttt{state == ATOMIC\_DO\_SYSCALL}\\
        \texttt{\&\& m.msg\_len != -1}%
    }%
}
\shortcut{swap}{%
    \texttt{state \& 0x7F == ATOMIC\_SWAP}%
}

\end{tikzpicture}
