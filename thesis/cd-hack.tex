% Replace Open Sans with Noto Sans.
\AtBeginDocument{
    % Load "Noto Sans" and define `\notosans`.
    \usepackage[
        nott,      % do not load not monospace
        sf,        % only load sans-serif
        sfdefault, % activate sans-serif
        % numerals
        lining,
        tabular,
        % light font
        light,
        semibold,
    ]{noto}
    % Overwrite tudscr's internal font macro `tud@cdfont@fam@lf` with `\notosans`.
    % \tempa = tud@x@fontspec@fam@*\tud@cdfont@fam@lf
    \edef\tempa{tud@x@fontspec@fam@\csname tud@cdfont@fam@lf\endcsname }%
    % \tempa = *\csname *\tempa \endcsname
    \expandafter\expandafter
    \expandafter\def
    \expandafter\expandafter
    \expandafter\tempa
    \expandafter\expandafter
    \expandafter{%
    \expandafter\csname
    \tempa      \endcsname
    }%
    % *\tempa = \notosans
    \expandafter\let\tempa\notosans
}
% Title page header (\faculty, ...) font.
\makeatletter
\let \old@tud@head@font@set \tud@head@font@set
\renewcommand*\tud@head@font@set{%
    {%
        \edef\tud@cdfont@fam@lf{\notosansfamily}%
        \old@tud@head@font@set
    }%
}
\makeatother
